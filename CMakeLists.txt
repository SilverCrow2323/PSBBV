cmake_minimum_required(VERSION 3.16)
project(PSBBV LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(psbbv_navigation STATIC src/app/navigation.cpp)
target_include_directories(psbbv_navigation PUBLIC include)
target_compile_options(psbbv_navigation PRIVATE -Wall -Wextra -Wpedantic)

option(PSBBV_BUILD_HOST_TESTS "Build portable navigation tests" ON)
if(PSBBV_BUILD_HOST_TESTS AND NOT CMAKE_CROSSCOMPILING)
    enable_testing()
    add_executable(psbbv_navigation_tests tests/navigation_tests.cpp)
    target_link_libraries(psbbv_navigation_tests PRIVATE psbbv_navigation)
    target_compile_options(psbbv_navigation_tests PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME psbbv_navigation COMMAND psbbv_navigation_tests)
endif()

if(DEFINED ENV{VITASDK})
    include("$ENV{VITASDK}/share/vita.cmake")
    add_executable(PSBBV src/main.cpp)
    target_link_libraries(PSBBV PRIVATE psbbv_navigation vita2d SceCtrl_stub SceLibKernel_stub SceDisplay_stub SceGxm_stub SceSharedFb_stub SceAppMgr_stub SceFont_stub SceCommonDialog_stub)
    target_compile_options(PSBBV PRIVATE -Wall -Wextra -Wpedantic)
    vita_create_self(PSBBV.self PSBBV)
    vita_create_vpk(PSBBV.vpk SCBV00001 PSBBV.self VERSION 01.00 NAME PSBBV)
else()
    message(STATUS "VitaSDK not detected: building the portable navigation core and host tests only.")
endif()
